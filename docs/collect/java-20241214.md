## PriorityQueue 优先队列 （最大优先队列和最小优先队列）

```java
package com.example.demo.datastructure;

import java.util.PriorityQueue;
import java.util.Comparator;

public class Solution {
    public static void main(String[] args) {
        // 创建一个默认排序的优先队列
        PriorityQueue<Integer> naturalOrderQueue = new PriorityQueue<>();
        naturalOrderQueue.add(5);
        naturalOrderQueue.add(3);
        naturalOrderQueue.add(8);
        System.out.println("默认排序的优先队列: " + naturalOrderQueue);
        while (!naturalOrderQueue.isEmpty()) {
            System.out.println(naturalOrderQueue.poll());
        }

        // 创建一个自定义排序的优先队列
        PriorityQueue<Integer> customOrderQueue = new PriorityQueue<>(Comparator.reverseOrder());
        customOrderQueue.add(5);
        customOrderQueue.add(3);
        customOrderQueue.add(8);
        System.out.println("自定义排序的优先队列: " + customOrderQueue);
        while (!customOrderQueue.isEmpty()) {
            System.out.println(customOrderQueue.poll());
        }

        // 使用自定义对象和比较器
        PriorityQueue<Customer> customerQueue = new PriorityQueue<>(Comparator.comparingInt(Customer::getId));
        customerQueue.add(new Customer(3, "Alice"));
        customerQueue.add(new Customer(1, "Bob"));
        customerQueue.add(new Customer(2, "Charlie"));
        System.out.println("自定义对象排序的优先队列: " + customerQueue);
        while (!customerQueue.isEmpty()) {
            Customer customer = customerQueue.poll();
            System.out.println(customer);
        }
    }
}

class Customer {
    private int id;
    private String name;

    public Customer(int id, String name) {
        this.id = id;
        this.name = name;
    }

    public int getId() {
        return id;
    }

    @Override
    public String toString() {
        return "Customer{id=" + id + ", name='" + name + "'}";
    }
}
```

```java
PriorityQueue<long[]> v = new PriorityQueue<>((x, y) -> {
    if (x[0] != y[0]) {
        return Long.compare(x[0], y[0]);
    } else {
        return Long.compare(x[1], y[1]);
    }
```
